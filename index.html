আসসালামু আলাইকুম ওয়া রাহমাতুল্লাহ। Hasaan Bazaar-এর শুভ সূচনা হয়েছে হালাল রুজি অন্বেষণ এবং রাসুলুল্লাহ (সা.)-এর সুন্নাহ ভিত্তিক মার্জিত জীবনধারার উপাদান প্রতিটি ধর্মপ্রাণ পরিবারের কাছে সর্বোচ্চ সততা ও আমানতদারিতার সাথে পৌঁছে দেওয়ার পবিত্র ব্রত নিয়ে। আমাদের প্রতিটি আতর, কটন পাঞ্জাবি, ঐতিহ্যবাহী টুপি এবং প্রাকৃতিক চাকের খাঁটি মধু আমি নিজে গুণগত মান যাচাই করে সরাসরি নির্ভরযোগ্য সোর্স থেকে সংগ্রহ করি। গ্রাহকদের অটুট আস্থা ও দোয়াই আমাদের মূল শক্তি।

Supabase টেবিল তৈরির SQL স্ক্রিপ্ট (এক ক্লিকে কপি করুন)

আপনার Supabase প্রজেক্টে ঢুকে বাম পাশের SQL Editor-এ গিয়ে নিচের কোডটি পেস্ট করে Run চাপুন। স্বয়ংক্রিয়ভাবে সব টেবিল ও পলিসি সক্রিয় হয়ে যাবে:

-- ১. প্রোডাক্ট টেবিল তৈরি
CREATE TABLE IF NOT EXISTS public.products (
    id BIGINT PRIMARY KEY,
    name TEXT NOT NULL,
    category TEXT NOT NULL,
    category_name TEXT NOT NULL,
    price NUMERIC NOT NULL,
    original_price NUMERIC,
    unit TEXT DEFAULT 'টি',
    rating NUMERIC DEFAULT 5.0,
    reviews INTEGER DEFAULT 0,
    badge TEXT,
    image TEXT NOT NULL,
    description TEXT,
    features JSONB DEFAULT '[]'::jsonb,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ২. অর্ডার টেবিল তৈরি
CREATE TABLE IF NOT EXISTS public.orders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    order_id TEXT UNIQUE NOT NULL,
    customer_name TEXT NOT NULL,
    customer_phone TEXT NOT NULL,
    customer_address TEXT NOT NULL,
    delivery_zone TEXT NOT NULL,
    payment_method TEXT NOT NULL,
    transaction_id TEXT,
    items JSONB NOT NULL,
    subtotal NUMERIC NOT NULL,
    delivery_fee NUMERIC NOT NULL,
    total NUMERIC NOT NULL,
    status TEXT DEFAULT 'pending',
    status_history JSONB DEFAULT '[]'::jsonb,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ৩. ইনডেক্সিং
CREATE INDEX IF NOT EXISTS idx_orders_order_id ON public.orders(order_id);
CREATE INDEX IF NOT EXISTS idx_orders_customer_phone ON public.orders(customer_phone);

-- ৪. Row Level Security (RLS) ও পলিসি
ALTER TABLE public.products ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.orders ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public can view active products" ON public.products FOR SELECT USING (is_active = true);
CREATE POLICY "Public can manage products" ON public.products FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public can create orders" ON public.orders FOR INSERT WITH CHECK (true);
CREATE POLICY "Public can track orders" ON public.orders FOR SELECT USING (true);
CREATE POLICY "Public can update orders" ON public.orders FOR UPDATE USING (true) WITH CHECK (true);

-- ৫. রিয়েলটাইম ব্রডকাস্ট চালু
ALTER PUBLICATION supabase_realtime ADD TABLE public.orders;
ALTER PUBLICATION supabase_realtime ADD TABLE public.products;
